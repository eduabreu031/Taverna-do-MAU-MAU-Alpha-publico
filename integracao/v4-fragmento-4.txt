<script id="online-v84-controller">
(()=>{

 const winner=players.find(p=>p.id===g.winner)?.name;finishPanel.querySelector('#onlineFinishTitle').textContent=winner?'Vitória de '+winner+'!':'Partida encerrada';
 const host=uid===hostId,ready=players.length>=2&&players.every(p=>p.online);
 finishText.textContent=host?(ready?'A mesa está pronta para uma nova partida.':'Aguarde todos se reconectarem para jogar novamente.'):'Aguarde o anfitrião iniciar uma nova partida ou saia da sala.';
 rematchBtn.classList.toggle('hidden',!host);rematchBtn.disabled=!host||!ready;

const st=el('onlineStatusText'),notice=el('onlineNotice'),lobby=el('onlineLobby');

function send(data){if(ws?.readyState!==WebSocket.OPEN){message('Conexão indisponível. Aguarde reconexão.');return false}ws.send(JSON.stringify(data));return true;}

function connect(code,existingToken=''){
 if(ws){ws.onclose=null;ws.close();ws=null;}
 clearTimeout(reconnectTimer);closing=false;room=code;token=existingToken;setStatus('Conectando à sala '+code+'…');

  const canStart=uid===hostId&&players.length>=2&&players.every(p=>p.online)&&!m.game;
  el('onlineStart').disabled=!canStart;el('onlineStart').classList.toggle('hidden',!!m.game);
  el('onlineStartHint').textContent=m.game?'Partida iniciada.':canStart?'A sala está pronta para começar.':uid===hostId?'Aguarde pelo menos dois participantes conectados.':'Aguarde o anfitrião iniciar.';
  if(m.game){
   if(game&&m.game.revision<revision&&!(game.phase==='finished'&&m.game.phase==='playing'))return;
   const appeared=!game, eventIds=m.events||[];
   const fresh=firstGameState?(eventIds.filter(e=>e.type==='start').slice(-1)):(eventIds.filter(e=>e?.id&&!lastEventIds.has(e.id)));
   for(const e of eventIds)if(e?.id)lastEventIds.add(e.id);firstGameState=false;
   if(lastEventIds.size>200)lastEventIds=new Set([...lastEventIds].slice(-160));
   if(game&&m.game.revision!==revision)comboReset();game=m.game;revision=game.revision;
   try{online=true;el('sceneGame').classList.add('online-live');window.__tavOriginalOnline.apply(game,players,uid,{opa,deadline:m.turnDeadline,serverNow:m.serverNow});}catch(e){online=false;console.error(e);message('Falha ao abrir a mesa: '+e.message);return;}
   const roundTransition=fresh.some(e=>e.type==='elimination')&&fresh.some(e=>e.type==='round_start');
   const ids=game.players.map(x=>x.id),ix=ids.indexOf(uid),relativeIds=ids.slice(ix).concat(ids.slice(0,ix));
   for(const ev of fresh){const playFx=()=>{try{window.__tavOriginalOnline.showOnlineEvent(ev,players,relativeIds);}catch(err){console.warn('Efeito online',err);}};
    if(roundTransition&&ev.type==='round_start')setTimeout(()=>{if(online&&window.__tavOriginalOnline.active)playFx();},2500);
    else playFx();}
   comboUpdate();if(m.opa){el('opaBtn').disabled=m.opa.playerId===uid;}
   else el('opaBtn').disabled=false;
   if(m.turnDeadline&&game.turnPlayerId===uid){const secs=Math.max(0,Math.ceil((m.turnDeadline-m.serverNow)/1000));el('clockText').textContent=secs+'s';}
   if(game.phase==='finished'||game.phase==='game_over'){setStatus('Partida encerrada');}showFinished(game);
  }else if(online){online=false;finishPanel.classList.add('hidden');el('sceneGame').classList.remove('online-live');window.__tavOriginalOnline.reset();}
  if(!online)setStatus(players.filter(p=>p.online).length+' participante(s) conectado(s)');
 };
 socket.onerror=()=>{setStatus('Sem conexão ao servidor');if(!online)message('Não foi possível conectar à sala.');};
 socket.onclose=e=>{
  if(closing||ws!==socket)return;
  setStatus('Conexão interrompida');
  if(e.code===4000||e.code===1008||retries>=5){message('Sessão encerrada. Volte ao menu e tente novamente.');return;}
  const pause=Math.min(12000,1200*2**retries++);reconnectTimer=setTimeout(()=>connect(code,store.get(key(code))||token),pause);
 };
}
const action=(action,cardId=null,suit=null)=>{if(!game||game.phase!=='playing')return;const payload={type:'action',action,revision:game.revision};if(cardId)payload.cardId=cardId;if(suit)payload.suit=suit;send(payload);};
const comboBtn=document.createElement('button');
comboBtn.type='button';comboBtn.id='onlineComboBtn';comboBtn.className='engraved-btn slim hidden';
comboBtn.textContent='COMBO';
el('mauBtn').after(comboBtn);
let comboMode=false,comboIds=[];
function comboReset(){comboMode=false;comboIds=[];comboBtn.textContent='COMBO';document.querySelectorAll('#humanHand [data-card]').forEach(b=>b.classList.remove('selected'));}
function comboUpdate(){
 const alive=game?.players?.filter(x=>x.alive!==false)?.length||0;
 const show=online&&game?.phase==='playing'&&alive===2&&game.turnPlayerId===uid;
 comboBtn.classList.toggle('hidden',!show);
 if(!show){comboReset();return;}
 comboBtn.textContent=comboMode?'JOGAR COMBO ('+comboIds.length+')':'COMBO';
 for(const b of document.querySelectorAll('#humanHand [data-card]'))b.classList.toggle('selected',comboIds.includes(b.dataset.card));
}
comboBtn.onclick=()=>{
 if(!online||!game||game.turnPlayerId!==uid)return;
 if(!comboMode){suppressClickUntil=0;comboMode=true;comboIds=[];message('Selecione duas ou mais cartas de efeito para o supercombo.');comboUpdate();return;}
 if(comboIds.length<2){message('Selecione pelo menos duas cartas para confirmar o supercombo.');return;}
 send({type:'action',action:'combo',revision:game.revision,cardId:[...comboIds]});comboReset();comboUpdate();
};
function comboSelect(id){if(!comboMode)return false;
 if(!game?.hand?.some(c=>c.id===id))return true;
 comboIds=comboIds.includes(id)?comboIds.filter(v=>v!==id):[...comboIds,id];comboUpdate();return true;

// Window capture is ordered BEFORE the original document-level gesture handler.
// This is strictly disabled outside an active online match.

function playCard(cardId){
 if(!game||game.turnPlayerId!==uid){message('Aguarde sua vez.');return;}
 const card=game.hand.find(c=>c.id===cardId);if(!card)return;
 window.__tavOriginalOnline.unlockAudio();
 if(card.rank==='J'&&game.hand.length>1){window.__tavOriginalOnline.chooseSuit(card.id,suit=>action('play',card.id,suit));return;}

}
function onlineTarget(e){return e.target?.closest?.('[data-card],#drawPile,#passBtn,#sevenBtn,#mauBtn,#opaBtn,#discardPile,#humanHand,#returnMenu')||null;}
function intercept(e){
 if(!online)return;
 const hit=onlineTarget(e);if(!hit)return;
 if(e.type==='click'&&performance.now()<suppressClickUntil&&(hit.matches?.('[data-card],#humanHand,#drawPile'))){e.stopImmediatePropagation();e.preventDefault();return;}
 if(e.type!=='click'&&e.type!=='dblclick'&&e.type!=='keydown')return;
 e.stopImmediatePropagation();e.preventDefault();
 if(e.type!=='click')return;
 const card=e.target.closest?.('[data-card]');
 if(hit.id==='returnMenu'){close(false);el('sceneWelcome').classList.remove('hidden');el('sceneMultiplayer').classList.add('hidden');return;}
 if(card){if(!comboSelect(card.dataset.card))playCard(card.dataset.card);}
 else if(hit.id==='drawPile')action('draw');
 else if(hit.id==='passBtn')action('pass');
 else if(hit.id==='sevenBtn')action('take_seven');
 else if(hit.id==='mauBtn')action('mau');
 else if(hit.id==='opaBtn'){if(!opa){message('Nenhum evento OPA!!! disponível.');return;}send({type:'accuse',eventId:opa.id});}
}
for(const t of ['click','dblclick','keydown'])window.addEventListener(t,intercept,true);
function removeGhost(){
 if(!drag)return;drag.card?.classList.remove('online-being-dragged');drag.ghost?.remove();el('discardPile')?.classList.remove('online-drop-target');drag=null;
}
function beginDrag(e){
 if(!online)return;
 const card=e.target?.closest?.('#humanHand [data-card]');
 const pile=e.target?.closest?.('#drawPile');
 if(!card&&!pile)return;
 // Prevent local offline gesture recognizer from getting an active touch/drag.
 e.stopImmediatePropagation();
 if(e.button!==undefined&&e.button!==0)return;
 drag={kind:card?'play':'draw',id:card?.dataset.card||null,card:card||pile,x:e.clientX,y:e.clientY,active:false,ghost:null};
}
function moveDrag(e){
 if(!online||!drag)return;
 e.stopImmediatePropagation();
 const distance=Math.hypot(e.clientX-drag.x,e.clientY-drag.y);
 if(!drag.active&&distance>12){
  drag.active=true;drag.ghost=document.createElement('div');drag.ghost.className='online-drag-ghost';drag.ghost.innerHTML=drag.card.querySelector('.big-card')?.outerHTML||'<div class="big-card card-back"><div class="coat">⚜</div></div>'; 
  document.body.append(drag.ghost);drag.card.classList.add('online-being-dragged');
 }
 if(!drag.active)return;
 e.preventDefault();drag.ghost.style.left=e.clientX+'px';drag.ghost.style.top=e.clientY+'px';
 const destination=drag.kind==='draw'?el('humanHand'):el('discardPile');
 const r=destination.getBoundingClientRect();
 const drop=e.clientX>=r.left-35&&e.clientX<=r.right+35&&e.clientY>=r.top-35&&e.clientY<=r.bottom+35;
 el('discardPile').classList.toggle('online-drop-target',drop&&drag.kind==='play');
}
function finishDrag(e){
 if(!online||!drag)return;
 e.stopImmediatePropagation();
 const d=drag,active=d.active;
 const r=(d.kind==='draw'?el('humanHand'):el('discardPile')).getBoundingClientRect();
 const over=e.clientX>=r.left-35&&e.clientX<=r.right+35&&e.clientY>=r.top-35&&e.clientY<=r.bottom+35;
 removeGhost();
 if(active){suppressClickUntil=performance.now()+260;e.preventDefault();if(over){if(d.kind==='draw')action('draw');else if(!comboSelect(d.id))playCard(d.id);}}
}

window.addEventListener('pointermove',moveDrag,{capture:true,passive:false});

for(const t of ['dragstart','dragover','drop','dragend'])window.addEventListener(t,e=>{

},true);
